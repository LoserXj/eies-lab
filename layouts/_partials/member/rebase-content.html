{{/* Rebase only anchor href attributes in copied HTML. Tokenize quoted attributes so text, code examples, and other attributes stay untouched. */}} {{- $paper := .page -}} {{- $html := printf "%s" .html -}} {{- range findRE `(?is)"']|"[^"]*"|'[^']*')*>` $html -}} {{- $tag := . -}} {{- $rebased := "" -}} {{- range findRESubmatch `(?s)(\s+)([^\s=<>/"']+)(?:\s*=\s*(?:"([^"]*)"|'([^']*)'|([^\s>]+)))?|.` $tag -}} {{- $attribute := index . 0 -}} {{- if eq (lower (index . 2)) "href" -}} {{- $href := (index . 3) | default (index . 4) | default (index . 5) -}} {{- $href = htmlUnescape $href -}} {{- $url := partial "member/source-url" (dict "page" $paper "url" $href) -}} {{- if ne $url $href -}} {{- $attribute = printf "%shref=\"%s\"" (index . 1) (htmlEscape $url) -}} {{- end -}} {{- end -}} {{- $rebased = printf "%s%s" $rebased $attribute -}} {{- end -}} {{- $html = replace $html $tag $rebased -}} {{- end -}} {{- return ($html | safeHTML) -}}
